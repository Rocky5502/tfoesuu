\begin{tikzpicture}[
  box/.style={draw=black!28, rounded corners=2pt, fill=white,
              minimum height=1.72cm, text width=5.65cm, align=left,
              inner sep=7pt, font=\small},
  sparse/.style={box, draw=focalteal, line width=0.85pt, fill=focalteal!5},
  dense/.style={box, draw=risktone, line width=0.85pt, fill=risktone!4},
  flow/.style={->, line width=0.8pt, draw=black!55},
  restore/.style={->, line width=0.9pt, draw=focalteal},
  lab/.style={font=\scriptsize, text=black!65, fill=white, inner sep=1.5pt},
  note/.style={font=\footnotesize, text=black!68, align=center}
]
\node[sparse] (zero) {
  \textbf{Sparse replay witness}\\[2pt]
  $\tau=0$\\
  null mismatch mass vanishes\\[2pt]
  $Q=\Theta(d^{-1}\log(1/\delta))$
};
\node[sparse, right=0.72cm of zero] (matched) {
  \textbf{Matched-scale tolerance}\\[2pt]
  $\tau=\Theta(d)$\\
  background and alternative shrink together\\[2pt]
  $Q=\Theta(d^{-1}\log(1/\delta))$
};

\node[dense, below=0.70cm of zero] (fixed) {
  \textbf{Fixed positive tolerance}\\[2pt]
  $\tau>0$ fixed as $d\to0$\\
  one-edge / high--low-only information is $\Theta(d^2)$\\[2pt]
  $Q=\Theta(d^{-2}\log(1/\delta))$
};
\node[sparse, right=0.72cm of fixed] (control) {
  \textbf{Complete control coverage}\\[2pt]
  fixed $\tau>0$, every candidate fractionally covered by matched controls\\[2pt]
  information becomes $\Theta(d)$\\
  $Q=\Theta(C_{\rm ctrl}d^{-1}\log(1/\delta))$
};

\draw[flow] (zero) -- node[lab,above] {$\tau$ grows with $d$} (matched);
\draw[flow] (matched.south west) -- node[lab,sloped,above] {fixed background mass} (fixed.north east);
\draw[restore] (fixed) -- node[lab,above,text=focalteal] {add matched replay control} (control);

\node[note, below=0.43cm of fixed, xshift=3.18cm, text width=12.2cm] {
  P18: $\displaystyle Q_{\rm thr}(\tau,d,\delta)
  =\Theta((\tau+d)d^{-2}\log(1/\delta))$ on one edge.
  \qquad
  P19: complete control coverage restores the linear minimax order; mixed
  topologies currently have exact lower-program asymptotics only.
};
\end{tikzpicture}
\caption{Replay phase map for P18--P19.  A fixed positive fairness tolerance
creates an interior mismatch mass and returns one-edge testing to quadratic
local complexity.  That loss is not universal: under partial replay, a
complete fractional cover by same-background controls cancels the benign
disparity pathwise and restores sparse linear witnesses.  The diagram summarizes
proved threshold-Bernoulli topology-complete regimes and contains no empirical
estimates.}
\label{fig:threshold-replay-phase}
\end{figure}
